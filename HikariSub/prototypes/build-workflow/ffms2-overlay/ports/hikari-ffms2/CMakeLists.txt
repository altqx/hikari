cmake_minimum_required(VERSION 3.28)
project(HikariFFMS2 VERSION 5.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(FFMPEG REQUIRED)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/source-sha256.json" source_lock)
string(JSON source_count LENGTH "${source_lock}")
math(EXPR last_source "${source_count}-1")
foreach(i RANGE ${last_source})
    string(JSON source_name MEMBER "${source_lock}" ${i})
    string(JSON expected GET "${source_lock}" "${source_name}")
    file(SHA256 "${FFMS2_SOURCE_DIR}/${source_name}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Pinned fork source bytes changed: ${source_name}")
    endif()
endforeach()
if(NOT FFMPEG_VERSION VERSION_EQUAL "7.1.2")
    message(FATAL_ERROR "This bounded probe requires exactly FFmpeg 7.1.2, got '${FFMPEG_VERSION}'")
endif()
foreach(component libavcodec libavformat libavutil libswscale libswresample)
    if(NOT FFMPEG_${component}_FOUND)
        message(FATAL_ERROR "Missing required pinned component: ${component}")
    endif()
endforeach()
file(SHA512 "${FFMS2_ADDITIONAL_SOURCE}" extra_digest)
if(NOT extra_digest STREQUAL "881b43761897a1c75204de2249b91fa744e72a6e88dc6d7c94ccf6d472eba6869cea34700e469443f2de92c0ba472a64594ece974201e9cbbec4f4f290ca3e4f")
    message(FATAL_ERROR "Private API implementation hash differs")
endif()
file(READ "${FFMS2_ADDITIONAL_SOURCE}" extra)
foreach(pair "../../ffms2/include/ffms.h|ffms.h" "../../ffms2/src/core/indexing.h|indexing.h")
    string(REPLACE "|" ";" fields "${pair}")
    list(GET fields 0 old)
    list(GET fields 1 new)
    string(FIND "${extra}" "#include \"${old}\"" position)
    if(position LESS 0)
        message(FATAL_ERROR "Pinned include relocation anchor missing: ${old}")
    endif()
    string(REPLACE "#include \"${old}\"" "#include \"${new}\"" extra "${extra}")
endforeach()
# Only include paths change; private API behavior remains the pinned source.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/indexing_additional.cpp" "${extra}")
set(core_sources audiosource ffms filehandle indexing track utils videosource videoutils zipfile)
list(TRANSFORM core_sources PREPEND "${FFMS2_SOURCE_DIR}/src/core/")
list(TRANSFORM core_sources APPEND ".cpp")
add_library(hikari_ffms2 STATIC ${core_sources} "${CMAKE_CURRENT_BINARY_DIR}/indexing_additional.cpp")
set_target_properties(hikari_ffms2 PROPERTIES EXPORT_NAME ffms2 OUTPUT_NAME hikari-ffms2 POSITION_INDEPENDENT_CODE ON)
target_compile_features(hikari_ffms2 PUBLIC cxx_std_17)
target_compile_definitions(hikari_ffms2 PUBLIC FFMS_STATIC PRIVATE FFMS_EXPORTS __STDC_CONSTANT_MACROS _FILE_OFFSET_BITS=64)
if(MSVC)
    target_compile_definitions(hikari_ffms2 PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
target_include_directories(hikari_ffms2
    PUBLIC "$<BUILD_INTERFACE:${FFMS2_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:include>"
    PRIVATE "${FFMS2_SOURCE_DIR}/src/core" ${FFMPEG_INCLUDE_DIRS})
target_link_libraries(hikari_ffms2 PRIVATE "$<BUILD_INTERFACE:${FFMPEG_LIBRARIES}>" ZLIB::ZLIB Threads::Threads)
install(TARGETS hikari_ffms2 EXPORT HikariFFMS2Targets ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}")
install(FILES "${FFMS2_SOURCE_DIR}/include/ffms.h" "${FFMS2_SOURCE_DIR}/include/ffmscompat.h" DESTINATION include)
install(EXPORT HikariFFMS2Targets NAMESPACE HikariFFMS2:: DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/HikariFFMS2")
configure_package_config_file(HikariFFMS2Config.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/HikariFFMS2Config.cmake"
    INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/HikariFFMS2")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/HikariFFMS2ConfigVersion.cmake"
    VERSION 5.1.0 COMPATIBILITY ExactVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/HikariFFMS2Config.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/HikariFFMS2ConfigVersion.cmake" DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/HikariFFMS2")
