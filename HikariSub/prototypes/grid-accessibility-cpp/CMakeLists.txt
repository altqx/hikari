cmake_minimum_required(VERSION 3.24)
project(HikariGridAccessibilityProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
qt_add_executable(grid_accessibility main.cpp)
target_link_libraries(grid_accessibility PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2)
target_compile_definitions(grid_accessibility PRIVATE PROBE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
    target_compile_options(grid_accessibility PRIVATE /utf-8)
endif()
